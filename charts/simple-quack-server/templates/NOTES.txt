In-cluster clients (plain HTTP):

  LOAD quack;
  ATTACH 'quack:{{ include "simple-quack-server.fullname" . }}.{{ .Release.Namespace }}' AS lake
      (TOKEN '<QUACK_TOKEN from secret {{ .Values.auth.existingSecret }}>', DISABLE_SSL true);
  FROM lake.query('USE {{ .Values.env.ATTACH_ALIAS | default "lake" }}');  -- once per session
  SELECT * FROM lake.<table>;                          -- reads, INSERT, CTAS
  SELECT * FROM lake.query('MERGE INTO ...');          -- UPDATE/DELETE/ALTER/MERGE/time travel
